```lua
function ns:ResolveSuggestedSpellID(key)
    if type(key) ~= "string" then
        return nil
    end
    for _, suggested in ipairs(ns.SUGGESTED_BUFFS) do
        if suggested.key == key and suggested.getCDMSpellID then
            return suggested.getCDMSpellID()
        end
    end
    return nil
end
```
