Replace the current lines 3-12 (the old backfill loop) with:

```lua
function ns:InitBuffEngine()
    local CURRENT_SCHEMA_VERSION = 1
    local ver = ns.db.schemaVersion or 0

    if ver < 1 then
        -- v0 -> v1: Replace enabled/displayMode with section (D-01)
        -- enabled=false -> "hidden", displayMode="buff" -> "buffs", else -> "bars"
        for _, entry in pairs(ns.db.trackedBuffs) do
            if not entry.section then
                if entry.enabled == false then
                    entry.section = "hidden"
                elseif entry.displayMode == "buff" then
                    entry.section = "buffs"
                else
                    -- "bar", nil, or any unexpected value -> bars (safe default)
                    entry.section = "bars"
                end
            end
            -- Remove deprecated fields regardless of section presence (D-02)
            entry.enabled = nil
            entry.displayMode = nil
        end
        ns.db.schemaVersion = 1
        print("|cff00ccffTerribleBuffTracker|r: DB migrated to v1.")
    end
end
```

IMPORTANT: Do NOT expand the `if not TerribleBuffTrackerDB` block in Core.lua. The schemaVersion field being nil is handled by `ns.db.schemaVersion or 0`. Core.lua is READ-ONLY in this phase.
