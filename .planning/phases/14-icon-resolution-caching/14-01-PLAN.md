
    2. **Add ns.metaIcons cache (D-01).** Module-level, not persisted:

       ```lua
       -- D-01: Eager-cached at-rest icons for meta-slots. Populated by ns:RefreshMetaIcons
       -- (Phase 14, called from CDMTab StartPreview). Read by ns:GetAtRestMetaIcon and by
       ```

    3. **Add ns:RefreshMetaIcons (D-02, D-05, D-06, D-07, D-08, D-09).** One function, both meta-slots inline (Claude's discretion per CONTEXT — kept folded for symmetry and single combat gate):

       ```lua
       -- D-02/D-03: Single scan entry point. Called from CDMTab StartPreview only —
       -- no PLAYER_EQUIPMENT_CHANGED / BAG_UPDATE_DELAYED hooks (D-04, ICON-06 N/A).
       -- D-07: Combat-gated. If locked down, leave ns.metaIcons as-is (stale or nil).
       -- D-09: If a chosen itemID's icon is nil/uncached, store nil — display falls back
       -- to 134400. We do NOT recurse to the next list entry (per user: ?-icon, not Bloodlust).
       function ns:RefreshMetaIcons()
           if InCombatLockdown() then
               return -- D-07: skip scan, do NOT blank cache
           end

           -- Trinket scan (D-05): equipped slots 13/14, first match wins
           local trinketIcon = nil
           local trinketItemID = nil
           for _, slot in ipairs({ INVSLOT_TRINKET1, INVSLOT_TRINKET2 }) do
               local equipped = GetInventoryItemID("player", slot)
               if equipped and TRINKET_ITEM_IDS[equipped] then
                   trinketItemID = equipped
                   break
               end
           end
           -- D-08: Fallback to first CSV entry whose icon resolves
           if not trinketItemID then
               for _, itemID in ipairs(TRINKET_FALLBACK_ORDER) do
                   if C_Item.GetItemIconByID(itemID) then
                       trinketItemID = itemID
                       break
                   end
               end
           end
           if trinketItemID then
               trinketIcon = C_Item.GetItemIconByID(trinketItemID) -- D-09: may be nil, stored as-is
           end
           ns.metaIcons.trinket = trinketIcon

           -- Pot scan (D-06): bag iteration over POT_ITEM_IDS in CSV order, first count>0 wins
           local potIcon = nil
           local potItemID = nil
           for _, itemID in ipairs(POT_FALLBACK_ORDER) do
               if (C_Item.GetItemCount(itemID) or 0) > 0 then
                   potItemID = itemID
                   break
               end
           end
           -- D-08: Fallback to first CSV entry whose icon resolves
           if not potItemID then
               for _, itemID in ipairs(POT_FALLBACK_ORDER) do
                   if C_Item.GetItemIconByID(itemID) then
                       potItemID = itemID
                       break
                   end
               end
           end
           if potItemID then
               potIcon = C_Item.GetItemIconByID(potItemID) -- D-09
           end
           ns.metaIcons.pot = potIcon
       end
       ```

    4. **Add ns:GetAtRestMetaIcon helper (D-10).** Single read path used by Display and CDMTab:

       ```lua
       -- D-10: Public read accessor. Returns 134400 (?-icon) when cache is nil, matching
       -- the GetSpellIcon fallback pattern and user's explicit fallback preference.
       function ns:GetAtRestMetaIcon(key)
           return ns.metaIcons[key] or 134400
       end
       ```

    5. **Wire SUGGESTED_BUFFS getCDMIcon (D-11).** In place, replace the two `getCDMIcon = nil` lines:
